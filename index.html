Pengaturan Integration Endpoint Webhook (Google Apps Script)

Hubungkan SIUS v8.0 Enterprise ke Google Spreadsheet via GAS WebApp API untuk sinkronisasi hasil ujian otomatis

Pastikan Web App di-deploy dengan akses "Anyone" (Siapa saja) pada Google Apps Script.
Fitur Otomatisasi Terintegrasi:
  • Mengirim Laporan Ujian (NIS, Nama, Mapel, Skor, Pelanggaran) secara otomatis saat siswa mengumpulkan.
  • Mengirim rekam jejak Audit Log keamanan secara terpusat.
  • Mode offline fallback: Jika koneksi GAS terputus, data tetap tersimpan aman di LocalStorage browser.
Script Google Apps Script (.gs) 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if(data.action === "PING") {
      return ContentService.createTextOutput(JSON.stringify({status:"SUCCESS", message:"GAS Active"})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if(data.action === "SUBMIT_EXAM") {
      sheet.appendRow([
        new Date(),
        data.nis,
        data.nama,
        data.kelas,
        data.mapel,
        data.skor_auto,
        data.skor_essay || 0,
        data.skor_final,
        data.jumlah_pelanggaran,
        data.status_penyelesaian
      ]);
      return ContentService.createTextOutput(JSON.stringify({status:"SUCCESS"})).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:"ERROR", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
